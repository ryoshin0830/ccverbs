# ccverbs v0.2.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cut the apply flow to two questions, move language/mode/scope into a `ccverbs config` settings TUI, and localise every user-facing string across five locales with detection that works on a Japanese Mac.

**Architecture:** A new `src/i18n/` holds one TypeScript catalog per locale with `en` as the type source, so a missing translation is a compile error; a new `src/config/` owns `~/.ccverbs/` and the relocated cache. The Ink app splits into a two-step main flow (`set` → `confirm`) and a separate settings app, both built from a shared `ChoiceScreen`. Help text becomes data so column alignment cannot drift across languages.

**Tech Stack:** TypeScript (ESM), Node >= 18, tsup, vitest, zod, Ink 6 + React 19.

**Spec:** `docs/superpowers/specs/2026-08-20-ccverbs-i18n-wizard-design.md`

## Global Constraints

- Version becomes `0.2.0`. No breaking changes: the `spinnerVerbs` shape written to `settings.json` is identical to 0.1.0, and no existing `--json` field is renamed or removed.
- Locales, exactly five: `en`, `ja`, `zh-Hans`, `zh-Hant`, `ko`. `en` is the fallback and the type source.
- `meta.reviewed` is `true` for `en` and `ja` only; `false` for `zh-Hans`, `zh-Hant`, `ko`.
- Config lives at `~/.ccverbs/config.json`; cache moves to `~/.ccverbs/cache/index.json` from `~/.cache/ccverbs/index.json`.
- Config defaults: `{ version: 1, language: "auto", mode: "replace", scope: "user" }`.
- Locale detection order: `--lang` → `CCVERBS_LANG` → config → `LC_ALL`/`LC_MESSAGES`/`LANG`/`LANGUAGE` → OS → `Intl` → `en`.
- **`C`, `POSIX`, and empty are "no preference", never English.** This is the bug the whole detection chain exists to avoid.
- OS locale query: `execFileSync`, `timeout: 500`, `stdio: ["ignore","pipe","ignore"]`, all failures non-fatal, skipped entirely when `CCVERBS_NO_OS_LOCALE=1` or when an earlier rung already decided.
- Catalogs ship inside the npm package. i18n must never depend on the network, because the "registry unavailable" message itself must be translated.
- The main wizard never writes config. Only `ccverbs config` writes config.
- Reading config never throws: corrupt or unknown values fall back to defaults and return warnings.
- `--json` output is always ordered by set `id`, independent of locale. Only human output is grouped by language.
- Exit codes unchanged: `0` ok, `1` runtime error, `2` usage error, `3` set not found, `4` registry unavailable.

```ts
// src/i18n/locales.ts
export type SupportedLocale = "en" | "ja" | "zh-Hans" | "zh-Hant" | "ko";

export const SUPPORTED_LOCALES = ["en", "ja", "zh-Hans", "zh-Hant", "ko"] as const;

/** Locales not yet reviewed by a native speaker. See spec section 3.6. */
export const UNREVIEWED_LOCALES = ["zh-Hans", "zh-Hant", "ko"] as const;

export function isSupportedLocale(value: unknown): value is SupportedLocale {
  return typeof value === "string" && (SUPPORTED_LOCALES as readonly string[]).includes(value);
}

/** Regions that use traditional Chinese characters. */
const HANT_REGIONS = new Set(["tw", "hk", "mo"]);

export function negotiate(tag: string): SupportedLocale | null {
  // Normalise: underscores to hyphens, drop the .encoding and @modifier suffixes.
  const cleaned = tag.trim().replace(/_/g, "-").split("@")[0]?.split(".")[0]?.toLowerCase();
  if (!cleaned) return null;

  const parts = cleaned.split("-").filter(Boolean);
  const language = parts[0];
  if (!language) return null;

  // "C" and "POSIX" mean "no locale configured" — not English.
  if (language === "c" || language === "posix") return null;

  if (language === "en") return "en";
  if (language === "ja") return "ja";
  if (language === "ko") return "ko";

  if (language === "zh") {
    const script = parts.find((p) => p === "hans" || p === "hant");
    if (script === "hant") return "zh-Hant";
    if (script === "hans") return "zh-Hans";
    const region = parts.slice(1).find((p) => p.length === 2);
    if (region && HANT_REGIONS.has(region)) return "zh-Hant";
    return "zh-Hans";
  }

  return null;
}
```

```ts
// src/i18n/os.ts
import { execFileSync } from "node:child_process";

function defaultRun(cmd: string, args: string[]): string {
  return execFileSync(cmd, args, {
    encoding: "utf8",
    timeout: 500,
    stdio: ["ignore", "pipe", "ignore"],
  });
}

/** A quoted string inside a macOS plist array. */
const PLIST_ENTRY = /"([^"]+)"/g;
const BARE_TAG = /^[A-Za-z]{2,3}(-[A-Za-z0-9]{2,8})*$/;

export function queryOsLocales(
  deps: { platform?: NodeJS.Platform; run?: (cmd: string, args: string[]) => string } = {},
): string[] {
  const platform = deps.platform ?? process.platform;
  const run = deps.run ?? defaultRun;

  try {
    if (platform === "darwin") {
      const out = run("defaults", ["read", "-g", "AppleLanguages"]);
      return [...out.matchAll(PLIST_ENTRY)].map((m) => m[1] as string);
    }
    if (platform === "win32") {
      const out = run("powershell", [
        "-NoProfile",
        "-Command",
        "Get-UICulture | Select-Object -ExpandProperty Name",
      ]);
      const tag = out.trim();
      return BARE_TAG.test(tag) ? [tag] : [];
    }
  } catch {
    // A missing binary, a timeout, or a non-zero exit must never break the CLI.
  }
  return [];
}
```

```ts
// src/i18n/resolve.ts
import { isSupportedLocale, negotiate, type SupportedLocale } from "./locales.js";
import { queryOsLocales } from "./os.js";

export type LocaleSource =
  | "flag" | "env" | "config" | "posix-env" | "os" | "intl" | "default";

export interface ResolveLocaleDeps {
  flagLang?: string | undefined;
  configLanguage?: string | undefined;
  env?: NodeJS.ProcessEnv;
  platform?: NodeJS.Platform;
  intlLocale?: string;
  queryOs?: () => string[];
}

const POSIX_VARS = ["LC_ALL", "LC_MESSAGES", "LANG", "LANGUAGE"] as const;

export function resolveLocale(deps: ResolveLocaleDeps = {}): {
  locale: SupportedLocale;
  source: LocaleSource;
} {
  const env = deps.env ?? process.env;
  const platform = deps.platform ?? process.platform;

  if (deps.flagLang) {
    const hit = negotiate(deps.flagLang);
    if (hit) return { locale: hit, source: "flag" };
  }

  const fromEnvVar = env.CCVERBS_LANG;
  if (fromEnvVar) {
    const hit = negotiate(fromEnvVar);
    if (hit) return { locale: hit, source: "env" };
  }

  if (deps.configLanguage && deps.configLanguage !== "auto") {
    if (isSupportedLocale(deps.configLanguage)) {
      return { locale: deps.configLanguage, source: "config" };
    }
  }

  for (const name of POSIX_VARS) {
    const value = env[name];
    if (!value) continue;
    const hit = negotiate(value);
    if (hit) return { locale: hit, source: "posix-env" };
  }

  if (!env.CCVERBS_NO_OS_LOCALE) {
    const tags = deps.queryOs ? deps.queryOs() : queryOsLocales({ platform });
    for (const tag of tags) {
      const hit = negotiate(tag);
      if (hit) return { locale: hit, source: "os" };
    }
  }

  const intlLocale =
    deps.intlLocale ?? Intl.DateTimeFormat().resolvedOptions().locale;
  const fromIntl = negotiate(intlLocale);
  if (fromIntl) return { locale: fromIntl, source: "intl" };

  return { locale: "en", source: "default" };
}
```

```ts
import { SUPPORTED_LOCALES, UNREVIEWED_LOCALES } from "../../src/i18n/locales.js";
import { COMMANDS, OPTIONS } from "../../src/help/model.js";

type Node = Record<string, unknown>;

/** Every leaf path with the arity of function leaves. */
function shape(node: Node, prefix = ""): Map<string, number | "string" | "boolean"> {
  const out = new Map<string, number | "string" | "boolean">();
  for (const [key, value] of Object.entries(node)) {
    const path = prefix ? `${prefix}.${key}` : key;
    if (typeof value === "function") out.set(path, value.length);
    else if (typeof value === "string") out.set(path, "string");
    else if (typeof value === "boolean") out.set(path, "boolean");
    else out.set(path, "string"), out.delete(path), shape(value as Node, path).forEach((v, k) => out.set(k, v));
  }
  return out;
}

const enShape = shape(getCatalog("en") as unknown as Node);

describe("catalogs", () => {

    for (const locale of SUPPORTED_LOCALES) {
```

```ts
// src/config/io.ts
import { mkdirSync, readFileSync, renameSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";
import { isSupportedLocale, type SupportedLocale } from "../i18n/locales.js";
import type { Scope } from "../settings/paths.js";

export interface CcverbsConfig {
  version: 1;
  language: "auto" | SupportedLocale;
  mode: "replace" | "append";
  scope: Scope;
}

export const DEFAULT_CONFIG: CcverbsConfig = {
  version: 1,
  language: "auto",
  mode: "replace",
  scope: "user",
};

const MODES = new Set(["replace", "append"]);
const SCOPES = new Set(["user", "project", "local"]);

export function readConfig(path: string): {
  config: CcverbsConfig;
  warnings: string[];
  existed: boolean;
} {
  let raw: string;
  try {
    raw = readFileSync(path, "utf8");
  } catch {
    return { config: { ...DEFAULT_CONFIG }, warnings: [], existed: false };
  }

  const warnings: string[] = [];
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    return {
      config: { ...DEFAULT_CONFIG },
      warnings: [`${path} is not valid JSON; using defaults`],
      existed: true,
    };
  }

  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    return {
      config: { ...DEFAULT_CONFIG },
      warnings: [`${path} is not a JSON object; using defaults`],
      existed: true,
    };
  }

  const input = parsed as Record<string, unknown>;
  const config = { ...DEFAULT_CONFIG };

  if (input.version !== 1) {
    warnings.push(`${path} has an unknown version; reading it as best I can`);
  }
  if (input.language === "auto" || isSupportedLocale(input.language)) {
    config.language = input.language as CcverbsConfig["language"];
  } else if (input.language !== undefined) {
    warnings.push(`ignoring an unknown language in ${path}`);
  }
  if (typeof input.mode === "string" && MODES.has(input.mode)) {
    config.mode = input.mode as CcverbsConfig["mode"];
  } else if (input.mode !== undefined) {
    warnings.push(`ignoring an unknown mode in ${path}`);
  }
  if (typeof input.scope === "string" && SCOPES.has(input.scope)) {
    config.scope = input.scope as Scope;
  } else if (input.scope !== undefined) {
    warnings.push(`ignoring an unknown scope in ${path}`);
  }

  return { config, warnings, existed: true };
}

export function writeConfig(path: string, config: CcverbsConfig): { warnings: string[] } {
  try {
    mkdirSync(dirname(path), { recursive: true });
    const tmp = `${path}.tmp`;
    writeFileSync(tmp, `${JSON.stringify(config, null, 2)}\n`, "utf8");
    renameSync(tmp, path);
    return { warnings: [] };
  } catch (error) {
    // Failing to remember a preference is less serious than failing to apply a
    // verb set, so this is a warning rather than an error.
    return { warnings: [`could not save settings to ${path}: ${(error as Error).message}`] };
  }
}
```

```ts
// src/config/migrate.ts
import { copyFileSync, existsSync, mkdirSync, readdirSync, renameSync, rmSync } from "node:fs";
import { dirname } from "node:path";
import { cachePath, legacyCachePath } from "./paths.js";

/** Moves the 0.1.0 cache under ~/.ccverbs. Safe to call on every startup. */
export function migrateCache(home?: string): { moved: boolean } {
  try {
    const next = cachePath(home);
    if (existsSync(next)) return { moved: false };

    const legacy = legacyCachePath(home);
    if (!existsSync(legacy)) return { moved: false };

    mkdirSync(dirname(next), { recursive: true });
    try {
      renameSync(legacy, next);
    } catch {
      copyFileSync(legacy, next);
      rmSync(legacy, { force: true });
    }

    const legacyDir = dirname(legacy);
    if (readdirSync(legacyDir).length === 0) rmSync(legacyDir, { recursive: true, force: true });

    return { moved: true };
  } catch {
    // A cache is disposable; losing it costs one refetch.
    return { moved: false };
  }
```

```ts
// src/help/render.ts
import type { Catalog } from "../i18n/en.js";
import { layoutWidth } from "../registry/schema.js";
import { COMMANDS, OPTIONS } from "./model.js";

const INDENT = "  ";
const GAP = 2;

function pad(text: string, width: number): string {
  return text + " ".repeat(Math.max(0, width - layoutWidth(text)));
}

function table(rows: { left: string; right: string }[]): string[] {
  const width = Math.max(...rows.map((r) => layoutWidth(r.left)));
  return rows.map((r) => `${INDENT}${pad(r.left, width + GAP)}${r.right}`);
}

function commandLeft(c: (typeof COMMANDS)[number]): string {
  return "arg" in c && c.arg ? `${c.name} <${c.arg}>` : c.name;
}

function optionLeft(o: (typeof OPTIONS)[number]): string {
  const short = "short" in o && o.short ? `-${o.short}, ` : "    ";
  const value = "value" in o && o.value ? ` <${o.value}>` : "";
  return `${short}--${o.long}${value}`;
}

export function renderHelp(t: Catalog): string {
  const lines: string[] = [
    `${t.common.appName} - ${t.help.tagline}`,
    "",
    t.help.usage,
    "",
    `${INDENT}${pad(t.common.appName, 24)}${t.help.defaultLine}`,
    "",
    t.help.commandsHeading,
    ...table(COMMANDS.map((c) => ({ left: commandLeft(c), right: t.help.commands[c.name] }))),
    "",
    t.help.optionsHeading,
    ...table(OPTIONS.map((o) => ({ left: optionLeft(o), right: t.help.options[o.long] }))),
    "",
    t.help.examplesHeading,
    ...table(t.help.examples.map((e) => ({ left: e.cmd, right: e.text }))),
    "",
    t.help.exitCodes,
    "",
    t.help.footer,
    "",
  ];
  return lines.join("\n");
}
```

```ts
  const { io, t } = deps;

  if (options.configKey === "reset") {
    const { warnings } = writeConfig(deps.configPath, { ...DEFAULT_CONFIG });
    return report(options, deps, warnings);
  }

  if (options.configKey) {
    const next = { ...deps.config, [options.configKey]: options.configValue } as CcverbsConfig;
    const { warnings } = writeConfig(deps.configPath, next);
    return report(options, deps, warnings, next);
  }

  // No key: show the table. The TUI path is chosen in cli.ts, not here.
  if (options.json) {
    io.out(JSON.stringify({ ok: true, ...describeConfig(deps) }));
    return EXIT.OK;
  }
  for (const line of renderConfigTable(deps)) io.out(line);
  return EXIT.OK;
}
```

The three helpers `runConfig` uses, all in `src/commands/config.ts`:

```ts
/** The --json payload from spec 10.2. */
function describeConfig(deps: CommandDeps): {
  language: { value: SupportedLocale; source: LocaleSource; explicit: boolean };
  mode: { value: "replace" | "append"; source: "config" | "default" };
  scope: { value: Scope; source: "config" | "default" };
  supportedLocales: readonly SupportedLocale[];
  unreviewedLocales: readonly SupportedLocale[];
  configPath: string;
  cachePath: string;
  cacheAgeMs: number | null;
  warnings: string[];
};

/** The human table: one row per setting, each with a source label. */
function renderConfigTable(deps: CommandDeps): string[];

/** Prints the result of a write, honouring --json, and returns the exit code. */
function report(
  options: Options,
  deps: CommandDeps,
  warnings: string[],
  next?: CcverbsConfig,
): number;
```

`language.explicit` is `true` when `source` is `"flag"`, `"env"` or `"config"`.
`mode.source` and `scope.source` are `"config"` when the config file existed and
carried that key, else `"default"` — so a write that silently failed is visible.
`report` returns `EXIT.OK` even when `warnings` is non-empty, because failing to
save a preference is not a failed command; it emits the warnings to stderr.

```tsx
import { Box, Text } from "ink";
import { useInput } from "ink";
import { useState } from "react";
import { layoutWidth } from "../../registry/schema.js";

export interface Choice<T> {
  value: T;
  label: string;
  hint?: string;
  note?: string;
}

export interface ChoiceScreenProps<T> {
  title: string;
  choices: Choice<T>[];
  initialValue: T;
  footer: string;
  onSelect: (value: T) => void;
  onBack: () => void;
}

export function ChoiceScreen<T>({
  title, choices, initialValue, footer, onSelect, onBack,
}: ChoiceScreenProps<T>) {
  const start = Math.max(0, choices.findIndex((c) => c.value === initialValue));
  const [index, setIndex] = useState(start);
  const width = Math.max(...choices.map((c) => layoutWidth(c.label)));

  useInput((input, key) => {
    if (key.escape) return onBack();
    if (key.upArrow) return setIndex((i) => Math.max(0, i - 1));
    if (key.downArrow) return setIndex((i) => Math.min(choices.length - 1, i + 1));
    if (key.return) {
      const chosen = choices[index];
      if (chosen) onSelect(chosen.value);
    }
  });

  return (
    <Box flexDirection="column">
      <Text bold>{title}</Text>
      <Box marginTop={1} flexDirection="column">
        {choices.map((c, i) => (
          <Text key={String(c.value)} inverse={i === index}>
            {i === index ? "> " : "  "}
            {c.label}
            {" ".repeat(Math.max(0, width - layoutWidth(c.label)) + 4)}
            <Text dimColor={i !== index}>
              {c.hint ?? ""}
              {c.note ? `   ${c.note}` : ""}
            </Text>
          </Text>
        ))}
      </Box>
      <Box marginTop={1}>
        <Text dimColor>{footer}</Text>
      </Box>
    </Box>
  );
}
```
